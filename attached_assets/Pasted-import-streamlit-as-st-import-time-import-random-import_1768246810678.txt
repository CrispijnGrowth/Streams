# --- FILE PATHS FOR SAVING DATA ---
def get_filename(username):
    # Sanitize username to be safe for filenames
    safe_name = "".join(x for x in username if x.isalnum())
    return f"progress_{safe_name}.json"

def save_data(username, data):
    # We remove image objects before saving text data to JSON
    # Images cannot be easily saved in a simple JSON file
    data_to_save = {k: v for k, v in data.items() if not k.endswith('_img')}
    with open(get_filename(username), 'w') as f:
        json.dump(data_to_save, f)

def load_data(username):
    filename = get_filename(username)
    if os.path.exists(filename):
        with open(filename, 'r') as f:
            return json.load(f)
    return None

def celebrate():
    msg = random.choice(txt["payoff"])

    st.session_state.step += 1
    st.rerun()
